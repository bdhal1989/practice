📘 BRIDGE PATTERN - EXAMPLES WITH IMPLEMENTATION
=================================================

DEFINITION

Bridge Pattern is a Structural Design Pattern
that separates Abstraction from Implementation
so both can change independently.

📘 BRIDGE PATTERN - COMPLETE IMPLEMENTATION NOTES
=================================================

DEFINITION

Bridge Pattern is a Structural Design Pattern
that separates Abstraction from Implementation
so that both can evolve independently.

Formula:

Abstraction
      HAS-A
Implementation

=================================================
EXAMPLE 1 - REMOTE + DEVICE
=================================================

Abstraction
------------
Remote

Implementation
------------
Device

interface Device {

    void on();

    void off();
}

class TV implements Device {

    public void on() {
        System.out.println("TV ON");
    }

    public void off() {
        System.out.println("TV OFF");
    }
}

class Radio implements Device {

    public void on() {
        System.out.println("Radio ON");
    }

    public void off() {
        System.out.println("Radio OFF");
    }
}

abstract class Remote {

    protected Device device;

    public Remote(Device device) {
        this.device = device;
    }

    abstract void power();
}

class BasicRemote extends Remote {

    public BasicRemote(Device device) {
        super(device);
    }

    void power() {
        device.on();
    }
}

Usage

Remote remote =
        new BasicRemote(
                new TV());

remote.power();

EXAMPLE 2 - SHAPE + COLOR
=================================================

Abstraction
------------
Shape

Implementation
------------
Color

interface Color {

    String fillColor();
}

class RedColor implements Color {

    public String fillColor() {
        return "Red";
    }
}

class BlueColor implements Color {

    public String fillColor() {
        return "Blue";
    }
}

abstract class Shape {

    protected Color color;

    public Shape(Color color) {
        this.color = color;
    }

    abstract void draw();
}

class Circle extends Shape {

    public Circle(Color color) {
        super(color);
    }

    void draw() {
        System.out.println(
            color.fillColor()
            + " Circle");
    }
}

Usage

Shape shape =
    new Circle(
        new RedColor());

shape.draw();

EXAMPLE 3 - NOTIFICATION + CHANNEL
=================================================

Abstraction
------------
Notification

Implementation
------------
MessageSender

interface MessageSender {

    void send(String message);
}

class EmailSender
implements MessageSender {

    public void send(String message) {

        System.out.println(
            "Email : "
            + message);
    }
}

class SmsSender
implements MessageSender {

    public void send(String message) {

        System.out.println(
            "SMS : "
            + message);
    }
}

abstract class Notification {

    protected MessageSender sender;

    public Notification(
            MessageSender sender) {

        this.sender = sender;
    }

    abstract void notifyUser(
            String message);
}

class AlertNotification
extends Notification {

    public AlertNotification(
            MessageSender sender) {

        super(sender);
    }

    void notifyUser(
            String message) {

        sender.send(message);
    }
}

Usage

Notification notification =
    new AlertNotification(
        new EmailSender());

notification.notifyUser(
    "Server Down");

EXAMPLE 7 - FOOD PLATFORM + DELIVERY PARTNER
=================================================

Abstraction
------------
FoodPlatform

Implementation
------------
DeliveryPartner

interface DeliveryPartner {

    void deliver(
        String item);
}

class Dunzo
implements DeliveryPartner {

    public void deliver(
            String item) {

        System.out.println(
            "Dunzo Delivered "
            + item);
    }
}

class Porter
implements DeliveryPartner {

    public void deliver(
            String item) {

        System.out.println(
            "Porter Delivered "
            + item);
    }
}

abstract class FoodPlatform {

    protected DeliveryPartner
            partner;

    public FoodPlatform(
            DeliveryPartner partner) {

        this.partner = partner;
    }

    abstract void placeOrder(
            String food);
}

class Swiggy
extends FoodPlatform {

    public Swiggy(
            DeliveryPartner partner) {

        super(partner);
    }

    void placeOrder(
            String food) {

        partner.deliver(food);
    }
}

Usage

FoodPlatform swiggy =
    new Swiggy(
        new Dunzo());

swiggy.placeOrder(
    "Biryani");
